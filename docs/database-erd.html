        string city

        datetime created_at
        datetime updated_at
    }

    Brand {
        uuid id PK
        uuid tenant_id
        string name
        boolean is_active
        datetime created_at
    }

    Product {

        uuid tenant_id
        uuid brand_id FK

        string colorway
        int msrp_price

        datetime created_at
        datetime updated_at
    }

    ProductVariant {
        uuid id PK
        uuid product_id FK
        string sku UK
        decimal size
        datetime created_at
    }

    Inventory {
        uuid id PK
        uuid tenant_id
        uuid variant_id FK
        uuid location_id FK

        int reserved_quantity
        int current_price
        datetime last_sold_at
        datetime last_markdown_at
        datetime received_at
        datetime updated_at
    }

    Sale {
        uuid id PK
        uuid tenant_id
        uuid variant_id FK
        uuid location_id FK
        uuid sold_by FK
        int quantity

        int total_price
        datetime sold_at
    }

    Transfer {
        uuid id PK
        uuid tenant_id
        uuid source_location_id FK
        uuid dest_location_id FK
        enum status "PENDING | APPROVED | REJECTED | IN_TRANSIT | COMPLETED | CANCELLED"
        uuid requested_by FK
        uuid reviewed_by FK
        string rejection_reason
        string note
        datetime created_at
        datetime updated_at
    }

    TransferItem {
        uuid id PK
        uuid transfer_id FK
        uuid variant_id FK
        int quantity
    }

    Reservation {
        uuid id PK
        uuid tenant_id
        uuid variant_id FK
        uuid location_id FK
        uuid reserved_by FK
        int quantity
        enum status "ACTIVE | CONVERTED | CANCELLED | EXPIRED"
        string customer_name
        string customer_phone
        datetime expires_at
        datetime created_at
        datetime updated_at
    }

    PriceHistory {
        uuid id PK
        uuid tenant_id
        uuid variant_id FK
        uuid location_id FK
        int old_price
        int new_price
        string reason
        datetime applied_at
    }

        uuid id PK
        uuid tenant_id
        uuid user_id FK
        string action
        string entity
        uuid entity_id
        json old_value
        json new_value
        datetime created_at
    }
